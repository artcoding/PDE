    \section{Equity Options}\label{sec:eq_opt}

    We start with the standard Black-Scholes PDE:
    \begin{equation}\label{eq:bs_pde_S}
        U_t + \mathcal{L}_S(r, q, \sigma)U = U_t + \frac{\sigma^2 S^2}{2} U_{SS} + (r - q) S\, U_S - r U = 0
    \end{equation}
    We immediately switch to the log stock price $x = \ln (S/S_0)$, and reverse time $\tau = T -t$.

    \begin{eqnarray}
        U_S &=& U_x / S \nonumber\\
        U_{SS} &=& (U_{xx} - U_x) / S^2\\
        U_t &=& -U_\tau \nonumber
    \end{eqnarray}

    In the new coordinates, the PDE \eqref{eq:bs_pde_S} becomes:
    \begin{eqnarray}
        U_\tau &=& \mathcal{L}_x(r, q, \sigma)\, U \label{eq:bs_pde}\\
        &=& \frac{\sigma^2}{2} U_{xx} + \left(r - q - \frac{\sigma^2}{2}\right) U_x - r U.\label{eq:bs_pde_detail}
        % U(x, 0) &=& \max[I(S_0 e^x - K), 0],
    \end{eqnarray}
    % where $I=1(-1)$ for Call(Put).

    \section{Solving the PDE}\label{sec:solving_pde}
    We now discretise equation \eqref{eq:bs_pde} along the time axis in equal intervals of size $k$, $\tau_j = j\, k$,
    where $j = 0, 1, \ldots, N$ and $T=k\, N$.
    We approximate the time derivative using a forward difference:
    \begin{equation}
        U_{\tau_j} \approx \frac{U^{j+1} - U^j}{k}
    \end{equation}
    where $U^j = U(x, \tau_j)$.
    
    We will need two discretization schemes for the spatial derivatives: the explicit Euler scheme and the implicit Crank-Nicolson.
    The former will be applied for the first few (four) steps to avoid unstable greeks due to the singularity of the second derivative
    at $\tau = 0$. The latter will be used for the rest of the time steps to ensure stability and accuracy.

    First, it is useful to rewrite the linear operator in a generic form:
    \begin{equation}\label{eq:generic_pde}
        \mathcal{L}_x\, U = a_0\, U + a_1\, U_x + a_2\, U_{xx},
    \end{equation}
    where $a_2 = 0.5\, \sigma^2$, $a_1 = r - q - a_2$, and $a_0 = -r$.

    We now discretize the spatial dimension into $M$ intervals of size $h$, $x_i = x_{\min} + i h$, where $i = 0, 1, \ldots, M$.
    We define $U_i^j = U(x_i, \tau_j)$, and

    \begin{eqnarray}
        U_{x} & \approx & \frac{1}{2h} \left(U_{i+1}^j - U_{i-1}^j\right) \label{eq:ders}\\
        U_{xx} & \approx & \frac{1}{h^2} \left(U_{i+1}^j - 2 U_i^j + U_{i-1}^j\right)\nonumber
    \end{eqnarray}

    We can now define completely the discretized operator $\mathcal{L}_x$:

    \begin{equation}\label{eq:operator}
        \mathcal{L}_x\, U_i^j = \left(\frac{a_2}{h^2} + \frac{a_1}{2h}\right) U_{i+1}^j + \left(a_0 - \frac{2a_2}{h^2}\right) U_i^j + \left(\frac{a_2}{h^2} - \frac{a_1}{2h}\right) U_{i-1}^j
    \end{equation}

        \subsection{Euler Scheme}\label{sec:euler_scheme}
        In this scheme, equation \eqref{eq:generic_pde} is evaluated at time step $\tau_j$.
        Therefore, the solution of \eqref{eq:bs_pde} at time $\tau_{j+1}$ is simply:
        \begin{eqnarray}
            U_i^{j+1} &\approx& U_i^j + k\, \mathcal{L}_x\, U_i^j \nonumber\\
            &\approx& k \left(\frac{a_2}{h^2} + \frac{a_1}{2h}\right) U_{i+1}^j + \left[1 + k\left(a_0 - \frac{2a_2}{h^2}\right)\right] U_i^j + k \left(\frac{a_2}{h^2} - \frac{a_1}{2h}\right) U_{i-1}^j
        \end{eqnarray}

        \subsection{Crank-Nicolson Method}\label{sec:cn_method}
        In the Crank-Nicolson scheme, we place the operator $\mathcal{L}_x$ at the midpoint between time steps $\tau_j$ and $\tau_{j+1}$.
        Equation \eqref{eq:bs_pde} is then approximated as a linear system of equations for $U^{j+1}$:
        \begin{eqnarray}
            \frac{U^{j+1} - U^j}{k} &=& \frac 1 2 \mathcal{L}_x (U^{j+1} + U^j)\nonumber\\
            \left(\mathcal{I} - \frac k  2 \mathcal{L}_x\right) U^{j+1} &=& \left(\mathcal{I} + \frac k  2 \mathcal{L}_x\right) U^j\label{eq:op_form}
        \end{eqnarray}

        Using the operator definition \eqref{eq:operator}, we get a tri-diagonal system of equations with:
        \begin{equation}
            \text{lower}\cdot U_{i-1}^{j+1} + \text{diag}\cdot U_i^{j+1} + \text{upper}\cdot U_{i+1}^{j+1} = \text{rhs}_i^j,
        \end{equation}
        where the coefficients are defined as follows:
        \begin{eqnarray}
            \text{diag} &=& 1 + \frac{k\, a_2}{h^2} - \frac{k\, a_0}{2} \nonumber\\
            \text{lower} &=& -\frac{k}{2} \left(\frac {a_2}{h^2} - \frac{a_1}{2h} \right)\\
            \text{upper} &=& -\frac{k}{2} \left(\frac {a_2}{h^2} + \frac{a_1}{2h} \right)\nonumber\\
            \text{rhs}_i^j &=& - \text{lower}\cdot U_{i-1}^j + (2 - \text{diag})\cdot U_i^j - \text{upper}\cdot U_{i+1}^j\nonumber
        \end{eqnarray}

        \subsection{Boundary Conditions}

        Solving the PDE requires two types of boundary conditions: the payoff function at maturity, and the conditions at the edges of the spatial domain. 
        Depending on the option type, we have the following payoff functions at maturity.

        \textbf{European Call}:
        \begin{eqnarray}
            U(x_i, 0) &=& U_{i}^0 = \max \left[S_0 \exp(x_i) - K, 0\right]\nonumber\\
            U(x_0, \tau_j) & = & U_0^j= 0,\\
            U(x_N, \tau_j) & = & U_{N}^j = S_0\, \exp(x_N - q\, \tau_j) - K\, \exp(-r\, \tau_j).\nonumber
        \end{eqnarray}

        \textbf{European Put}:
        \begin{eqnarray}
            U(x_i, 0) &=& U_{i}^0 = \max \left[K - S_0 \exp(x_i), 0\right]\nonumber\\
            U(x_0, \tau_j) & = & U_0^j =  K\, \exp(-r\, \tau_j) - S_0\, \exp(x_0 - q\, \tau_j),\\
            U(x_N, \tau_j) & = & U_{N}^j = 0.\nonumber
        \end{eqnarray}

        \textbf{American Call}:
        \begin{eqnarray}
            U(x_i, 0) &=& U_{i}^0 = \max \left[S_0 \exp(x_i) - K, 0\right]\nonumber\\
            U(x_0, \tau_j) & = & U_0^j= 0,\\
            U(x_i, \tau_j) & = & U_i^j = \ge S_0 \exp(x_i) - K.\nonumber
        \end{eqnarray}
        
        \textbf{American Put}:
        \begin{eqnarray}
            U(x_i, 0) &=& U_{i}^0 = \max \left[K - S_0 \exp(x_i), 0\right]\nonumber\\
            U(x_N, \tau_j) & = & U_{N}^j = 0,\\
            U(x_i, \tau_j) & = & U_i^j = \ge K - S_0 \exp(x_i).\nonumber
        \end{eqnarray}

        The inequality conditions simply mean that at each time steps we need to check if the solution is below
        the intrinsic value, and if so, replace it with the intrinsic value.

    \section{Greeks by PDE}

    \subsection{Vega}
    Differentiate the original PDE \eqref{eq:bs_pde_S} with respect to $\sigma$ to get the PDE for Vega, and define
    $v = \partial U / \partial \sigma$:
    \begin{equation}
        \sigma S^2 U_{SS} + v_t + \mathcal{L}_S v = 0
    \end{equation}
    Using the same change of variables as before, we get the PDE for $v$ in terms of $U$:
    \begin{eqnarray}
        \mathcal{L}_x v - v_\tau &=& \sigma (U_x - U_{xx}),\nonumber\\
        v(x, \tau = 0) &=& 0,\\
        v(x_{\min, \max}, \tau) &=& 0.
    \end{eqnarray}
    Specifically, $U_{xx} (x_{\max(\min)}, \tau) = U_x (x_{\max(\min)}, \tau) = I\, S_0 e^{x_{\max(\min)} -q \tau}$
    and $U_{xx}(x_{\min(\max)}, \tau) = U_x(x_{\min(\max)}, \tau) = 0$. Hence $U_x - U_{xx} = 0$ at the boundaries.

    \subsection{Rho}
    Defining $\rho = \partial U / \partial r$, and differentiating \eqref{eq:bs_pde_S} with respect to $r$, we get:
    
    \begin{equation}
        S\, U_S - U + \rho_t + \mathcal{L}_S \rho = 0
    \end{equation}
    Using the same change of variables as before, we get the PDE for $\rho$ in terms of $U$:
    \begin{eqnarray}
        \mathcal{L}_x \rho - \rho_\tau &=& U - U_{x},\\
        \rho(x, \tau = 0) &=& 0,\\
        \rho(x_{\min(\max)}, \tau) &=& 0,\\
        \rho(x_{\max(\min)}, \tau) & = & I\, \tau\, K\, e^{-r \tau}.
    \end{eqnarray}
